import ctypes
import hashlib
import json
import os
import pathlib
import select
import shutil
import subprocess
import sys


root = pathlib.Path('/tmp/int3-pr33569-compare')
build = root / 'Release'
logs = root / 'logs'
source = pathlib.Path('/datadisks/disk1/jiafa/accuracy/ort-int3-pr33569-compare')
result_path = logs / 'build-result.json'
if not result_path.exists():
    matches = subprocess.run(['pgrep', '-f', '^python3 /tmp/int3-pr33569-compare-build.py$'], capture_output=True, text=True, check=False)
    for process_id in matches.stdout.split():
        try:
            descriptor = os.pidfd_open(int(process_id))
        except ProcessLookupError:
            continue
        print(f'Waiting for existing build driver PID {process_id}', flush=True)
        select.select([descriptor], [], [])
        os.close(descriptor)
build_result = json.loads(result_path.read_text())
print(json.dumps(build_result, indent=2), flush=True)
summary = {'head': '4ba6c38384cf5b12d4aa4570cd15af2461c4d515', 'source': str(source), 'build': str(build), 'build_result': build_result}
if build_result['returncode']:
    lines = (logs / 'build.log').read_text().splitlines()
    errors = [line for line in lines if any(token in line.lower() for token in ('error:', 'fatal error', 'failed:', 'ninja: build stopped'))]
    summary['errors'] = errors
    (logs / 'validation.json').write_text(json.dumps(summary, indent=2) + '\n')
    print('\n'.join(line[:2400] for line in errors), flush=True)
    raise SystemExit(build_result['returncode'])

libraries = sorted(path for path in build.glob('libonnxruntime*.so*') if path.is_file() and not path.is_symlink())
bindings = list(build.glob('**/onnxruntime_pybind11_state*.so'))
artifacts = {}
for path in libraries + bindings:
    artifacts[str(path)] = {'bytes': path.stat().st_size, 'sha256': hashlib.file_digest(path.open('rb'), 'sha256').hexdigest()}
    inspection = subprocess.run(['ldd', str(path)], capture_output=True, text=True)
    (logs / (path.name + '.ldd.log')).write_text(inspection.stdout + inspection.stderr)
summary['artifacts'] = artifacts
core = next(path for path in libraries if path.name.startswith('libonnxruntime.so'))
probes = {
    'core': f'import ctypes; library=ctypes.CDLL({str(core)!r}); library.OrtGetApiBase.restype=ctypes.c_void_p; assert library.OrtGetApiBase(); print("OrtGetApiBase OK")',
    'cuda': f'import ctypes; ctypes.CDLL({str(build / "libonnxruntime_providers_shared.so")!r}, mode=ctypes.RTLD_GLOBAL); ctypes.CDLL({str(build / "libonnxruntime_providers_cuda.so")!r}); print("CUDA provider dlopen OK")',
}
for name, code in probes.items():
    probe = subprocess.run([sys.executable, '-X', 'faulthandler', '-c', code], capture_output=True, text=True)
    (logs / f'{name}-load.log').write_text(probe.stdout + probe.stderr)
    summary[f'{name}_load_returncode'] = probe.returncode
    print(name, probe.returncode, probe.stdout, probe.stderr, flush=True)
if bindings:
    package = root / 'python-runtime' / 'onnxruntime'
    capi = package / 'capi'
    capi.mkdir(parents=True, exist_ok=True)
    shutil.copy2(source / 'onnxruntime/__init__.py', package / '__init__.py')
    for filename in ('__init__.py', '_ld_preload.py', 'onnxruntime_inference_collection.py', 'onnxruntime_validation.py'):
        shutil.copy2(source / 'onnxruntime/python' / filename, capi / filename)
    shutil.copy2(build / 'onnxruntime/capi/_pybind_state.py', capi / '_pybind_state.py')
    for path in libraries + bindings:
        destination = capi / path.name
        if not destination.exists():
            destination.symlink_to(path)
    python = '/tmp/qmoe-validation-venv/bin/python'
    environment = dict(os.environ, PYTHONPATH=str(package.parent))
    check = subprocess.run([python, '-c', 'import json,pathlib,onnxruntime as ort; from onnxruntime.capi import onnxruntime_pybind11_state as native; assert pathlib.Path(ort.__file__).is_relative_to("/tmp/int3-pr33569-compare/python-runtime"); print(json.dumps({"package":ort.__file__,"native":native.__file__,"version":ort.get_version_string(),"providers":ort.get_available_providers(),"build_info":ort.get_build_info()},indent=2))'], env=environment, cwd='/tmp', capture_output=True, text=True)
    (logs / 'python-import.log').write_text(check.stdout + check.stderr)
    summary['python_import_returncode'] = check.returncode
    summary['python_runtime'] = str(package.parent)
    print(check.stdout + check.stderr, flush=True)
(logs / 'validation.json').write_text(json.dumps(summary, indent=2) + '\n')
print(json.dumps(summary, indent=2), flush=True)
raise SystemExit(summary.get('python_import_returncode', 0))
