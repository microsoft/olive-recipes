import hashlib
import json
import subprocess
import sys


SOURCE = '/datadisks/disk1/jiafa/accuracy/ort-int3-pr2-pr3'
LOGS = '/tmp/int3-pr33555-compare/logs'
HEAD = 'd75ac16f377c0c783d289046f5cf2efe6bb5f9bc'


def git(*arguments, source=SOURCE):
    return subprocess.check_output(['git', '-C', source, *arguments], text=True)


def snapshot():
    return {'head': git('rev-parse', 'HEAD').strip(),
            'status': git('status', '--porcelain=v1', '--untracked-files=normal'),
            'tracked_diff_sha256': hashlib.sha256(subprocess.check_output(
                ['git', '-C', SOURCE, 'diff', 'HEAD', '--binary'])).hexdigest(),
            'untracked_smoke_sha256': subprocess.check_output([
                'sha256sum', SOURCE + '/docs/design/int3-qmoe-pr3-session-smoke.json'], text=True).split()[0],
            'onnx_head': git('rev-parse', 'HEAD', source='/datadisks/disk1/jiafa/accuracy/onnxruntime/cmake/external/onnx').strip(),
            'onnx_status': git('status', '--porcelain=v1', source='/datadisks/disk1/jiafa/accuracy/onnxruntime/cmake/external/onnx')}


def record(name, data):
    subprocess.run(['docker', 'exec', '-i', 'jiafa-dev', 'python3', '-c',
                    'import pathlib,sys; pathlib.Path(sys.argv[1]).write_text(sys.stdin.read())',
                    LOGS + '/' + name], input=json.dumps(data, indent=2) + '\n', text=True, check=True)


before = snapshot()
assert before['head'] == HEAD, before
assert before['status'] == '?? docs/design/int3-qmoe-pr3-session-smoke.json\n', before
record('source-before-completion.json', before)
record('completion-command.json', {'command': ['python3', __file__], 'wait_for_container_pid': 537291})
print('Waiting for existing build driver PID 537291, then staging and validating', flush=True)
wait_code = r'''import json,os,pathlib,select,subprocess,sys
logs=pathlib.Path('/tmp/int3-pr33555-compare/logs')
try:
    descriptor=os.pidfd_open(537291)
except ProcessLookupError:
    descriptor=None
if descriptor is not None:
    select.select([descriptor],[],[])
    os.close(descriptor)
result=json.loads((logs/'build-result.json').read_text())
print(json.dumps(result,indent=2),flush=True)
if result['returncode']:
    print('\n'.join((logs/'build.log').read_text().splitlines()[-50:]),flush=True)
    sys.exit(result['returncode'])
sys.exit(subprocess.call(['python3','/tmp/int3-pr33555-compare-driver.py','validate']))
'''
result = subprocess.run(['docker', 'exec', 'jiafa-dev', 'python3', '-c', wait_code])
after = snapshot()
record('source-after-completion.json', after)
record('completion-result.json', {'returncode': result.returncode, 'source_unchanged': before == after})
assert before == after, {'before': before, 'after': after}
print('Source HEAD, tracked diff, pre-existing untracked file, and ONNX dependency unchanged', flush=True)
raise SystemExit(result.returncode)
