import hashlib
import json
import os
import pathlib
import re
import shlex
import shutil
import subprocess
import sys
import time


ROOT = pathlib.Path('/tmp/int3-pr33555-compare')
LOGS = ROOT / 'logs'
BUILD = pathlib.Path('/tmp/int3-pr2-pr3-clean/Release')
SOURCE = pathlib.Path('/datadisks/disk1/jiafa/accuracy/ort-int3-pr2-pr3')
REFERENCE = pathlib.Path('/tmp/int3-pr33569-compare')
HEAD = 'd75ac16f377c0c783d289046f5cf2efe6bb5f9bc'
TARGETS = ['onnxruntime', 'onnxruntime_pybind11_state', 'onnxruntime_providers_cuda']


def save(name, data):
    (LOGS / name).write_text(json.dumps(data, indent=2) + '\n')


def cache(path):
    entries = {}
    for line in path.read_text().splitlines():
        match = re.fullmatch(r'([^#/:][^:]*):([^=]+)=(.*)', line)
        if match:
            entries[match[1]] = {'type': match[2], 'value': match[3]}
    return entries


def run(name, command, environment=None):
    (LOGS / f'{name}-command.sh').write_text(shlex.join(command) + '\n')
    print(shlex.join(command), flush=True)
    started = time.time()
    with (LOGS / f'{name}.log').open('w') as output:
        process = subprocess.Popen(command, stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                                   text=True, env=environment, cwd='/tmp')
        for line in process.stdout:
            output.write(line)
            output.flush()
            print(line, end='', flush=True)
        returncode = process.wait()
    save(f'{name}-result.json', {'returncode': returncode, 'elapsed_seconds': time.time() - started})
    if returncode:
        raise SystemExit(returncode)


def configure():
    LOGS.mkdir(parents=True, exist_ok=False)
    shutil.copy2(__file__, LOGS / 'driver.py')
    shutil.copy2(BUILD / 'CMakeCache.txt', LOGS / 'before-CMakeCache.txt')
    shutil.copy2(REFERENCE / 'Release/CMakeCache.txt', LOGS / 'reference-CMakeCache.txt')
    before = cache(LOGS / 'before-CMakeCache.txt')
    reference = cache(LOGS / 'reference-CMakeCache.txt')
    overrides = {key: entry for key, entry in before.items() if key.startswith('FETCHCONTENT_SOURCE_DIR_')}
    assert all(reference.get(key, {}).get('value') == entry['value'] for key, entry in overrides.items())
    assert all(pathlib.Path(entry['value']).is_dir() for entry in overrides.values())
    settings = dict(overrides)
    for key, entry in reference.items():
        if entry['type'] not in ('INTERNAL', 'STATIC') and key.startswith(('Python_', 'Python3_', 'PYBIND11_')):
            settings[key] = entry
    for key in ('onnxruntime_BUILD_SHARED_LIB', 'onnxruntime_ENABLE_PYTHON', 'onnxruntime_ENABLE_DLPACK'):
        settings[key] = reference[key]
    assert settings['onnxruntime_ENABLE_PYTHON']['value'] == 'ON'
    assert settings['onnxruntime_BUILD_SHARED_LIB']['value'] == 'ON'
    save('configure-settings.json', settings)
    command = ['cmake', '-S', str(SOURCE / 'cmake'), '-B', str(BUILD)]
    command.extend(f'-D{key}:{entry["type"]}={entry["value"]}' for key, entry in sorted(settings.items()))
    run('configure', command)
    current = cache(BUILD / 'CMakeCache.txt')
    assert current['CMAKE_HOME_DIRECTORY']['value'] == str(SOURCE / 'cmake')
    assert current['CMAKE_CUDA_ARCHITECTURES']['value'] == '80'
    assert all(current.get(key, {}).get('value') == entry['value'] for key, entry in settings.items())
    preserved = {key: entry for key, entry in before.items()
                 if entry['type'] not in ('INTERNAL', 'STATIC') and
                 (key.startswith(('onnxruntime_', 'FETCHCONTENT_')) or key in
                  ('CMAKE_CUDA_ARCHITECTURES', 'CMAKE_BUILD_TYPE', 'CMAKE_CUDA_COMPILER', 'CMAKE_CXX_COMPILER', 'CMAKE_C_COMPILER'))
                 and key not in settings}
    differences = {key: [entry, current.get(key)] for key, entry in preserved.items()
                   if current.get(key, {}).get('value') != entry['value']}
    assert not differences, differences
    shutil.copy2(BUILD / 'CMakeCache.txt', LOGS / 'configured-CMakeCache.txt')
    save('cache-validation.json', {'settings_match': True, 'dependency_source_overrides': overrides,
                                  'preserved_settings': preserved, 'differences': differences,
                                  'existing_build_reconfigured_in_place': True})
    print(f'Cache validation PASS: {len(overrides)} dependency source overrides preserved', flush=True)


def build():
    assert json.loads((LOGS / 'configure-result.json').read_text())['returncode'] == 0
    assert json.loads((LOGS / 'cache-validation.json').read_text())['settings_match']
    run('build', ['ninja', '-C', str(BUILD), '-j8', *TARGETS])


def validate():
    assert json.loads((LOGS / 'build-result.json').read_text())['returncode'] == 0
    libraries = sorted(path for path in BUILD.glob('libonnxruntime*.so*')
                       if path.is_file() and not path.is_symlink())
    binding = BUILD / 'onnxruntime_pybind11_state.so'
    assert binding.is_file()
    artifacts = {}
    for path in libraries + [binding]:
        with path.open('rb') as stream:
            digest = hashlib.file_digest(stream, 'sha256').hexdigest()
        artifacts[str(path)] = {'bytes': path.stat().st_size, 'sha256': digest}
        inspection = subprocess.run(['ldd', str(path)], capture_output=True, text=True)
        (LOGS / f'{path.name}.ldd.log').write_text(inspection.stdout + inspection.stderr)
    save('artifact-hashes.json', artifacts)
    package = ROOT / 'python-runtime/onnxruntime'
    capi = package / 'capi'
    capi.mkdir(parents=True, exist_ok=False)
    shutil.copy2(SOURCE / 'onnxruntime/__init__.py', package / '__init__.py')
    for name in ('__init__.py', '_ld_preload.py', 'onnxruntime_inference_collection.py', 'onnxruntime_validation.py'):
        shutil.copy2(SOURCE / 'onnxruntime/python' / name, capi / name)
    shutil.copy2(BUILD / 'onnxruntime/capi/_pybind_state.py', capi / '_pybind_state.py')
    for path in libraries + [binding]:
        (capi / path.name).symlink_to(path)
    environment = dict(os.environ, PYTHONPATH=str(package.parent), PYTHONDONTWRITEBYTECODE='1')
    python = cache(BUILD / 'CMakeCache.txt')['Python_EXECUTABLE']['value']
    probe = '''import json,pathlib,numpy as np,onnxruntime as ort
from onnxruntime.capi import onnxruntime_pybind11_state as native
root=pathlib.Path('/tmp/int3-pr33555-compare/python-runtime')
assert pathlib.Path(ort.__file__).is_relative_to(root)
assert pathlib.Path(native.__file__).is_relative_to(root)
providers=ort.get_available_providers()
assert 'CUDAExecutionProvider' in providers, providers
print(json.dumps({'package':ort.__file__,'native':native.__file__,'version':ort.get_version_string(),'providers':providers,'build_info':ort.get_build_info()},indent=2))
host=np.array([1.25],dtype=np.float32)
value=ort.OrtValue.ortvalue_from_numpy(host,'cuda',0)
assert value.device_name()=='cuda',value.device_name()
np.testing.assert_array_equal(value.numpy(),host)
print('CUDA OrtValue allocation and host round-trip PASS; no model/inference')
'''
    (LOGS / 'validate_runtime.py').write_text(probe)
    run('python-cuda-ortvalue', [python, '-X', 'faulthandler', str(LOGS / 'validate_runtime.py')], environment)
    manifest = {}
    for path in sorted(package.rglob('*')):
        if path.is_file():
            with path.open('rb') as stream:
                digest = hashlib.file_digest(stream, 'sha256').hexdigest()
            manifest[str(path.relative_to(package.parent))] = {'sha256': digest,
                       'symlink_target': str(path.readlink()) if path.is_symlink() else None}
    save('staging-manifest.json', manifest)
    save('provenance.json', {'pinned_head': HEAD, 'source': str(SOURCE), 'build': str(BUILD),
         'reference': str(REFERENCE), 'python_runtime': str(package.parent),
         'targets': TARGETS, 'model_benchmark_run': False, 'standalone_provider_ctypes_run': False,
         'cuda_validation': 'ORT CUDA OrtValue single float allocation and host round-trip',
         'staging': 'same source-copy and library-symlink layout as PR33569; not a wheel',
         'source_before': 'one pre-existing untracked docs/design/int3-qmoe-pr3-session-smoke.json; no tracked changes'})
    save('validation.json', {'status': 'PASS', 'head': HEAD, 'artifacts': artifacts,
         'python_import_returncode': 0, 'cuda_ortvalue_returncode': 0,
         'python_runtime': str(package.parent)})
    print('Build, staging, Python import, and ORT CUDA OrtValue validation PASS', flush=True)


if __name__ == '__main__':
    {'configure': configure, 'build': build, 'validate': validate}[sys.argv[1]]()
